cmake_minimum_required(VERSION 3.19)
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(power C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(power main.c)
target_compile_options(power PRIVATE -Wall -Wextra -Werror)
target_link_libraries(power PRIVATE pico_stdlib pico_stdio_usb pico_low_power hardware_pll)
pico_enable_stdio_uart(power 0)
pico_enable_stdio_usb(power 1)
if(SECURE_BOOT_PKEY)
    pico_sign_binary(power "${SECURE_BOOT_PKEY}")
    pico_hash_binary(power)
endif()
pico_add_extra_outputs(power)
